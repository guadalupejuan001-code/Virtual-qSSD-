cmake_minimum_required(VERSION 3.16)
project(Virtual_qSSD LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(Threads REQUIRED)
find_package(Git REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" DEPENDENCIES)
# The manifest is also checked by setup.py. CMake 3.16 supports the original
# projects; use Python for JSON parsing instead of raising its version minimum.
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_dependencies.py"
  RESULT_VARIABLE VERIFIED)
if(NOT VERIFIED EQUAL 0)
  message(FATAL_ERROR "Missing/unreviewed original cores; run python3 scripts/setup.py")
endif()
set(CPU "${CMAKE_CURRENT_SOURCE_DIR}/deps/reverse-pc")
set(ETHER "${CMAKE_CURRENT_SOURCE_DIR}/deps/mundo-x")
set(GPU "${CMAKE_CURRENT_SOURCE_DIR}/deps/gpu-virtual")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_gpu.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_gpu.py"
  "${GPU}/include" "${CMAKE_CURRENT_BINARY_DIR}/gpu-compat" RESULT_VARIABLE GPU_COMPAT)
if(NOT GPU_COMPAT EQUAL 0)
  message(FATAL_ERROR "GPU read-only interface verification failed")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CPU}/scripts/prepare-ether-compat.py"
  "${ETHER}" "${CMAKE_CURRENT_BINARY_DIR}/ether-compat" RESULT_VARIABLE COMPAT)
if(NOT COMPAT EQUAL 0)
  message(FATAL_ERROR "Original REVERSE-PC compatibility verification failed")
endif()
add_library(qssd INTERFACE)
target_include_directories(qssd INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/ether-compat"
  "${ETHER}/native/include" "${CPU}/src" "${CMAKE_CURRENT_BINARY_DIR}/gpu-compat" "${GPU}/include")
target_compile_definitions(qssd INTERFACE REVERSE_PC_HAVE_BOUNDED_PRODUCT_IMPORT=1)
target_link_libraries(qssd INTERFACE Threads::Threads)
add_executable(qssd_space src/main.cpp)
target_link_libraries(qssd_space PRIVATE qssd)
include(CTest)
if(BUILD_TESTING)
  add_executable(qssd_gpu_original_tests "${GPU}/tests/pruebas.cpp")
  target_include_directories(qssd_gpu_original_tests PRIVATE "${GPU}/tests")
  target_link_libraries(qssd_gpu_original_tests PRIVATE qssd)
  add_test(NAME qssd_original_gpu_equivalence COMMAND qssd_gpu_original_tests "${CMAKE_CURRENT_BINARY_DIR}/original-gpu-checkpoint.xgs")
  add_executable(qssd_tests tests/state_space.cpp)
  target_link_libraries(qssd_tests PRIVATE qssd)
  add_test(NAME qssd_shared_authority COMMAND qssd_tests)
  add_test(NAME qssd_checkpoint_cli COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py" "$<TARGET_FILE:qssd_space>")
  set_tests_properties(qssd_shared_authority qssd_checkpoint_cli PROPERTIES TIMEOUT 180)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(qssd INTERFACE -Wall -Wextra -Wpedantic)
endif()
